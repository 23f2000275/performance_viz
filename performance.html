Employee Department Distribution

This visualization shows the distribution of employees across departments and highlights the count for the Marketing department.

# Author: 23f2000275@ds.study.iitm.ac.in import pandas as pd import matplotlib.pyplot as plt import seaborn as sns # Sample employee dataset (first 10 rows for demonstration) from io import StringIO data = StringIO(""" employee_id,department,region,performance_score,years_experience,satisfaction_rating EMP001,R&D,Middle East,83.73,14,5 EMP002,R&D,Latin America,87.82,12,4.7 EMP003,IT,Middle East,84.86,4,4.1 EMP004,IT,Africa,76.63,1,3 EMP005,R&D,Asia Pacific,65.75,14,3.9 EMP006,Marketing,Europe,78.12,6,4.5 EMP007,Marketing,Europe,80.25,7,4.6 EMP008,Marketing,North America,72.30,3,3.8 EMP009,Sales,Asia Pacific,90.50,11,4.9 EMP010,Sales,Latin America,88.10,9,4.2 """) # Load the dataset df = pd.read_csv(data) # Calculate frequency count for the "Marketing" department marketing_count = (df['department'] == 'Marketing').sum() print(f'Frequency count for "Marketing" Department: {marketing_count}') # Create histogram of department distribution plt.figure(figsize=(8, 5)) sns.countplot(data=df, x='department', palette='Set2') plt.title('Department Distribution of Employees') plt.xlabel('Department') plt.ylabel('Count') plt.xticks(rotation=45) plt.tight_layout() plt.savefig("chart-llm.png", bbox_inches='tight') plt.close()
